function rollCaseItem(c,user){
  if(user&&!user.pityCounters)user.pityCounters={};
  const pity=user?(user.pityCounters[c.id]||0):0;
  if(user&&pity>=PITY_LIMIT){
    const winPool=ITEMS.filter(i=>i.price>=c.price*1.2&&i.price<=c.price*5);
    user.pityCounters[c.id]=0;
    if(winPool.length>0)return pick(winPool);
  }
  const w=c.w,total=Object.values(w).reduce((a,b)=>a+b,0);
  let r=Math.random()*total,rarity='consumer';
  for(const k in w){r-=w[k];if(r<=0){rarity=k;break}}
  const minVal=Math.floor(c.price*MIN_DROP_RATIO);
  let pool=[];
  for(let i=RARITY_ORDER.indexOf(rarity);i<RARITY_ORDER.length;i++){
    pool=ITEMS.filter(it=>it.rarity===RARITY_ORDER[i]&&it.price>=minVal);
    if(pool.length>0)break;
  }
  if(pool.length===0)pool=ITEMS.filter(it=>it.rarity===rarity);
  const item=pick(pool);
  if(user){
    if(item.price>=c.price)user.pityCounters[c.id]=0;
    else user.pityCounters[c.id]=pity+1;
  }
  return item;
}

function ensureEmergencyFunds(u){
  if(!u)return false;
  if(u.balance>0)return false;
  if(u.inventory&&u.inventory.length>0)return false;
  if(DB.market&&DB.market.some(l=>l.seller===u.nick))return false;
  if(DB.trades&&DB.trades.some(t=>t.status==='pending'&&t.from===u.nick&&((t.youGive.items&&t.youGive.items.length>0)||t.youGive.currency>0)))return false;
  if(now()-(u.emergencyLastGrant||0)<EMERGENCY_COOLDOWN_MS)return false;
  u.balance=50;u.emergencyLastGrant=now();
  toast('💸 Аварийный бонус: +50 валюты','ok');
  return true;
}

function validatePassword(pw){
  if(pw.length<8)return 'Пароль должен быть не менее 8 символов';
  if(/[а-яА-ЯёЁ]/.test(pw))return 'Только английские буквы, цифры и символы';
  if(!/[a-zA-Z]/.test(pw))return 'Пароль должен содержать английские буквы';
  if(!/[0-9]/.test(pw))return 'Пароль должен содержать цифры';
  const low=pw.toLowerCase();
  const easy=['12345678','87654321','password','qwerty12','qwertyui','abcdefgh','00000000','11111111','123456789','1234567890','qwerty123','admin123'];
  if(easy.some(e=>low.includes(e)))return 'Пароль слишком лёгкий';
  if(/^(.)\1+$/.test(pw))return 'Пароль слишком лёгкий';
  const seq='0123456789';
  if(seq.includes(low)||[...seq].reverse().join('').includes(low))return 'Пароль слишком лёгкий';
  let asc=true,desc=true;
  for(let i=1;i<pw.length;i++){if(pw.charCodeAt(i)!==pw.charCodeAt(i-1)+1)asc=false;if(pw.charCodeAt(i)!==pw.charCodeAt(i-1)-1)desc=false;}
  if(asc||desc)return 'Пароль слишком лёгкий';
  return null;
}

function doLogin(){
  const nick=$('#aNick').value.trim(),pw=$('#aPw').value,err=$('#aErr');
  if(!nick||!pw){err.textContent='Заполните все поля';return}
  const u=DB.users[nick];
  if(!u){err.textContent='Пользователь не найден';return}
  if(u.pw!==hash(pw)){err.textContent='Неверный пароль';return}
  if(isBanned(u)){err.textContent='Забанен: '+u.banned.reason;return}
  u.lastSeen=now();
  if(!u.loginHistory)u.loginHistory=[];
  u.loginHistory.unshift({date:now(),nick});u.loginHistory=u.loginHistory.slice(0,20);
  STATE.user=nick;STATE.screen='main';save();playClick();render();
}
function doRegister(){
  const nick=$('#aNick').value.trim(),pw=$('#aPw').value,pw2=$('#aPw2').value,err=$('#aErr');
  if(!nick||!pw){err.textContent='Заполните все поля';return}
  if(!/^[a-zA-Z0-9_]{3,20}$/.test(nick)){err.textContent='Ник: 3-20 символов (латиница/цифры/_)';return}
  if(DB.users[nick]){err.textContent='Ник занят. Попробуйте: '+suggestNick(nick);return}
  const v=validatePassword(pw);if(v){err.textContent=v;return}
  if(pw!==pw2){err.textContent='Пароли не совпадают';return}
  DB.users[nick]={nick,pw:hash(pw),balance:500,inventory:[],role:'user',friends:[],requests:[],openedCases:0,caseOpened:{},pityCounters:{},lastSeen:now(),created:now(),loginHistory:[{date:now(),nick}],lastVipDaily:'',banned:null};
  STATE.user=nick;STATE.screen='main';save();playWin();render();
}

function sendFriendRequest(nick){
  const u=cur();if(u.friends.includes(nick)){toast('Уже в друзьях','err');return}
  if(!DB.users[nick])return;
  if(!u.sentRequests)u.sentRequests=[];
  if(u.sentRequests.includes(nick)){toast('Уже отправлено','err');return}
  u.sentRequests.push(nick);
  if(!DB.users[nick].requests)DB.users[nick].requests=[];
  if(!DB.users[nick].requests.includes(u.nick))DB.users[nick].requests.push(u.nick);
  save();toast('Заявка отправлена','ok');searchPlayer();
}
function acceptFriend(nick){
  const u=cur(),other=DB.users[nick];if(!other)return;
  u.requests=(u.requests||[]).filter(n=>n!==nick);
  if(!u.friends.includes(nick))u.friends.push(nick);
  if(!other.friends)other.friends=[];
  if(!other.friends.includes(u.nick))other.friends.push(u.nick);
  other.sentRequests=(other.sentRequests||[]).filter(n=>n!==u.nick);
  save();toast('Теперь друзья','ok');render();
}
function declineFriend(nick){const u=cur();u.requests=(u.requests||[]).filter(n=>n!==nick);save();render()}
function removeFriend(nick){if(!confirm('Удалить '+nick+'?'))return;const u=cur();u.friends=u.friends.filter(n=>n!==nick);const o=DB.users[nick];if(o)o.friends=(o.friends||[]).filter(n=>n!==u.nick);save();toast('Удалён','info');render()}

function totalSourceValue(){
  const u=cur();if(!u)return 0;
  return STATE.upgrade.items.reduce((s,idx)=>{
    const id=u.inventory[idx];
    return s+(ITEM_BY_ID[id]?.price||0);
  },0);
}

function filterTargets(){
  const total=totalSourceValue();
  let targets=ITEMS.slice();
  if(total>0){
    targets=targets.filter(it=>{
      const ch=Math.floor(total/it.price*100);
      return ch>=1&&ch<=90;
    });
  }
  if(STATE.upgrade.pctFilter&&total>0){
    const preset=STATE.upgrade.pctFilter;
    targets=targets.filter(it=>{
      const ch=Math.floor(total/it.price*100);
      return ch>=1&&ch<=preset;
    });
  }
  if(STATE.upgrade.priceFilter){
    const pf=parseFloat(STATE.upgrade.priceFilter);
    if(!isNaN(pf)&&pf>0)targets=targets.filter(it=>it.price>=pf*0.7&&it.price<=pf*1.3);
  }
  if(STATE.upgrade.pctFilter&&total>0){
    targets.sort((a,b)=>{
      const ca=Math.floor(total/a.price*100);
      const cb=Math.floor(total/b.price*100);
      return cb-ca;
    });
  } else {
    targets.sort((a,b)=>a.price-b.price);
  }
  return targets.slice(0,40);
}

function toggleSourceItem(idx){
  if(STATE.upgradeAnimating)return;
  const u=cur();
  if(idx<0||idx>=u.inventory.length)return;
  const pos=STATE.upgrade.items.indexOf(idx);
  if(pos>=0)STATE.upgrade.items.splice(pos,1);
  else{
    if(STATE.upgrade.items.length>=5){toast('Максимум 5 предметов','err');return}
    STATE.upgrade.items.push(idx);
  }
  if(STATE.upgrade.target){
    const total=totalSourceValue();
    const t=ITEM_BY_ID[STATE.upgrade.target];
    if(!t||total===0)STATE.upgrade.target=null;
    else{
      const ch=Math.floor(total/t.price*100);
      if(ch<1||ch>90)STATE.upgrade.target=null;
    }
  }
  render();
}

function selectTarget(id){
  if(STATE.upgradeAnimating)return;
  if(!STATE.upgrade.items.length){toast('Сначала выберите предметы','err');return}
  const total=totalSourceValue();
  const t=ITEM_BY_ID[id];
  const ch=Math.floor(total/t.price*100);
  if(ch<1){toast('Шанс слишком мал (<1%)','err');return}
  if(ch>90){toast('Шанс превышает 90%','err');return}
  STATE.upgrade.target=id;
  render();
}

function doUpgrade(){
  if(STATE.upgradeAnimating)return;
  const u=cur();
  const sourceIndices=STATE.upgrade.items.slice();
  const targetId=STATE.upgrade.target;
  if(!sourceIndices.length||!targetId){toast('Выберите предметы и цель','err');return}
  const sourceIds=sourceIndices.map(idx=>u.inventory[idx]).filter(Boolean);
  if(sourceIds.length!==sourceIndices.length){toast('Предметы не найдены','err');return}
  const targetItem=ITEM_BY_ID[targetId];
  const totalVal=sourceIds.reduce((s,id)=>s+(ITEM_BY_ID[id]?.price||0),0);
  const chance=Math.min(90,Math.floor(totalVal/targetItem.price*100));
  if(chance<1){toast('Шанс слишком мал','err');return}
  if(chance>90){toast('Шанс превышает 90%','err');return}

  STATE.upgradeAnimating=true;
  const btn=$('#upBtn');if(btn)btn.disabled=true;
  // удаляем именно ВЫБРАННЫЕ индексы (а не все копии по id)
  const removeSet=new Set(sourceIndices);
  u.inventory=u.inventory.filter((_,i)=>!removeSet.has(i));
  save();

  const win=Math.random()*100<chance;
  const needle=$('#upNeedle');const center=$('#upCenter');
  needle.style.display='block';
  const greenEnd=(chance/100)*360;
  const finalAngle=win?greenEnd*0.1+Math.random()*greenEnd*0.8:greenEnd+(360-greenEnd)*0.1+Math.random()*(360-greenEnd)*0.8;
  const start=performance.now(),dur=3200;let lastBeep=0;

  function anim(t){
    const p=Math.min(1,(t-start)/dur);const ease=1-Math.pow(1-p,3);
    const angle=ease*(360*4+finalAngle);
    needle.style.transform=`translateX(-50%) rotate(${angle}deg)`;
    if(t-lastBeep>100-p*70){beep(600+Math.random()*200,.02,'sine',0.02);lastBeep=t}
    if(p<1)requestAnimationFrame(anim);
    else{
      if(win){
        u.inventory.push(targetId);
        const r=RARITIES[targetItem.rarity];
        center.innerHTML=`<div style="width:70px;height:70px">${weaponSVG(targetItem)}</div><div style="color:${r.color};font-size:11px;margin-top:4px;text-align:center;padding:0 8px">${targetItem.name}</div>`;
        playWin();toast('ПОБЕДА! '+targetItem.name,'ok');
      }else{
        const refund=Math.floor(totalVal*UPGRADE_REFUND_PCT);
        u.balance+=refund;
        center.innerHTML=`<div style="font-size:60px;color:#dc2626">❌</div><div style="color:#dc2626;font-size:12px;margin-top:4px">Проигрыш</div><div style="color:#fbbf24;font-size:11px;margin-top:4px">+${fmt(refund)} 💰 (4%)</div>`;
        playLose();toast(`Проигрыш. Возврат ${fmt(refund)} 💰 (4%)`,'err');
      }
      ensureEmergencyFunds(u);
      save();
      STATE.upgradeAnimating=false;
      setTimeout(()=>{
        STATE.upgrade.items=[];
        STATE.upgrade.target=null;
        render();
      },2200);
    }
  }
  requestAnimationFrame(anim);
}

  const items=indices.map(i=>u.inventory[i]).filter(Boolean);
  if(items.length===0&&curAmt===0){toast('Выберите предметы или валюту','err');return}
  DB.trades.push({id:'t'+now()+rnd(1,999),from:u.nick,to,youGive:{items,currency:curAmt},theyGive:{items:[],currency:0},status:'pending',created:now()});
  save();toast('Отправлено','ok');render();
}
function acceptTrade(id){
  const u=cur(),t=DB.trades.find(x=>x.id===id);if(!t||t.status!=='pending')return;
  const from=DB.users[t.from];if(!from){toast('Нет отправителя','err');return}
  if(t.youGive.currency>u.balance){toast('Недостаточно','err');return}
  for(const it of t.youGive.items)if(!u.inventory.includes(it)){toast('Нет предмета','err');return}
  for(const it of t.theyGive.items)if(!from.inventory.includes(it)){toast('У отправителя нет','err');return}
  // удаляем по одному (чтобы не снести все копии)
  const removeOne=(arr,id)=>{const i=arr.indexOf(id);if(i>=0)arr.splice(i,1)};
  t.youGive.items.forEach(id=>removeOne(u.inventory,id));
  t.theyGive.items.forEach(id=>removeOne(from.inventory,id));
  u.inventory.push(...t.theyGive.items);
  from.inventory.push(...t.youGive.items);
  u.balance-=t.youGive.currency||0;u.balance+=t.theyGive.currency||0;
  from.balance+=t.youGive.currency||0;from.balance-=t.theyGive.currency||0;
  t.status='accepted';ensureEmergencyFunds(u);ensureEmergencyFunds(from);save();toast('Трейд завершён','ok');playWin();render();
}
function declineTrade(id){const t=DB.trades.find(x=>x.id===id);if(t)t.status='declined';save();render()}

function changePassword(){
  const u=cur();const old=$('#pwOld').value,n1=$('#pwNew').value,n2=$('#pwNew2').value;
  if(hash(old)!==u.pw){toast('Неверный текущий пароль','err');return}

  const v=validatePassword(n1);if(v){toast(v,'err');return}
